cmake_minimum_required(VERSION 3.27)
project(sdk_cache_warmup LANGUAGES CXX)
find_package(Qt6 REQUIRED COMPONENTS Widgets Concurrent)
find_package(KF6CoreAddons REQUIRED)
find_package(KF6Config REQUIRED)
qt_standard_project_setup()
qt_add_executable(sdk-smoke main.cpp)
target_link_libraries(sdk-smoke PRIVATE Qt6::Widgets Qt6::Concurrent KF6::CoreAddons KF6::ConfigCore)
set_target_properties(sdk-smoke PROPERTIES QT_WASM_PTHREAD_POOL_SIZE 2)
target_link_options(sdk-smoke PRIVATE -sASSERTIONS=1)
