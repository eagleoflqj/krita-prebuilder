cmake_minimum_required(VERSION 3.27)
project(sdk_i18n LANGUAGES CXX)
find_package(Qt6 REQUIRED COMPONENTS Widgets Qml Concurrent)
find_package(KF6I18n REQUIRED)
qt_standard_project_setup()
get_target_property(configs KF6::I18n IMPORTED_CONFIGURATIONS)
foreach(config IN LISTS configs)
  string(TOUPPER "${config}" upper)
  set_property(TARGET KF6::I18n PROPERTY IMPORTED_LOCATION_${upper} "${I18N_PREFIX}/target/lib/libKF6I18n.a")
endforeach()
qt_add_executable(sdk-i18n main.cpp)
target_link_libraries(sdk-i18n PRIVATE Qt6::Widgets Qt6::Qml Qt6::Concurrent KF6::I18n
  "${I18N_PREFIX}/target/lib/libktranscript.a" "${I18N_PREFIX}/target/lib/libintl.a")
set_target_properties(sdk-i18n PROPERTIES QT_WASM_PTHREAD_POOL_SIZE 2)
target_link_options(sdk-i18n PRIVATE -sASSERTIONS=1 "--preload-file=${CATALOGS}@/probe/share")
