cmake_minimum_required(VERSION 3.22)
project(krita_event_loop_probe LANGUAGES CXX)
find_package(Qt6 6.10 REQUIRED COMPONENTS Widgets Concurrent)
qt_standard_project_setup()
qt_add_executable(loop-probe main.cpp)
target_link_libraries(loop-probe PRIVATE Qt6::Widgets Qt6::Concurrent)
set_target_properties(loop-probe PROPERTIES QT_WASM_PTHREAD_POOL_SIZE 2 QT_WASM_EXPORT_NAME kritaLoopProbe)
target_compile_options(loop-probe PRIVATE -pthread -fwasm-exceptions)
target_link_options(loop-probe PRIVATE -sJSPI=1 -fwasm-exceptions -sASSERTIONS=1)
